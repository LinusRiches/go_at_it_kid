        const player = new Image();
player.src = 'assets/goat.png';

let playerX = 300;
let playerY = 300;
const playerWidth = 64;
const playerHeight = 64;
const speed = 2;

function drawPlayer() {
ctx.imageSmoothingEnabled = false;
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.drawImage(player, playerX, playerY, playerWidth, playerHeight);
}
player.onload = () => {
  drawPlayer();
}});
function gameLoop() {
  ctx.clearRect(0, 0, canvas.width, canvas.height); // Clear the canvas
  drawPlayer(); // Draw the player
  requestAnimationFrame(gameLoop);
}
gameLoop();

document.addEventListener('keydown', (event) => {
  const speed = 5; // Movement speed
  if (event.key === 'ArrowUp') player.y -= speed; // Move up
  if (event.key === 'ArrowDown') player.y += speed; // Move down
  if (event.key === 'ArrowLeft') player.x -= speed; // Move left
  if (event.key === 'ArrowRight') player.x += speed; // Move right
});
